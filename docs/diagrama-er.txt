' =========================
' AUTH
' =========================
package "auth" {
  entity "roles" as roles {
    *id : UUID <<PK>>
    --
    *name : VARCHAR(30) <<UQ>>
    description : VARCHAR(120)
  }

  entity "users" as users {
    *id : UUID <<PK>>
    --
    *first_name : VARCHAR(80)
    *last_name : VARCHAR(80)
    *email : VARCHAR(120) <<UQ>>
    *password_hash : VARCHAR(255)
    *provider : VARCHAR(20)
    *status : VARCHAR(20)
    *role_id : UUID <<FK>>
    created_at : TIMESTAMP
    updated_at : TIMESTAMP
  }

  entity "password_reset_tokens" as password_reset_tokens {
    *id : UUID <<PK>>
    --
    *user_id : UUID <<FK>>
    *token : VARCHAR(255) <<UQ>>
    *expires_at : TIMESTAMP
    *used : BOOLEAN
    created_at : TIMESTAMP
  }

  entity "sessions" as sessions {
    *id : UUID <<PK>>
    --
    *user_id : UUID <<FK>>
    *refresh_token : VARCHAR(255) <<UQ>>
    *expires_at : TIMESTAMP
    revoked_at : TIMESTAMP
    created_at : TIMESTAMP
  }
}

package "profile" {
  entity "student_profiles" as student_profiles {
    *user_id : UUID <<PK, FK>>
    --
    school_name : VARCHAR(120)
    grade_level : VARCHAR(20)
    target_university : VARCHAR(120)
    target_career : VARCHAR(120)
    study_shift : VARCHAR(30)
    is_anonymous_allowed : BOOLEAN
  }

  entity "moderator_profiles" as moderator_profiles {
    *user_id : UUID <<PK, FK>>
    --
    permission_level : VARCHAR(30)
    active : BOOLEAN
  }

  entity "admin_profiles" as admin_profiles {
    *user_id : UUID <<PK, FK>>
    --
    department : VARCHAR(80)
  }

  entity "teacher_profiles" as teacher_profiles {
    *user_id : UUID <<PK, FK>>
    --
    specialty : VARCHAR(120)
    verification_status : VARCHAR(20)
    bio : TEXT
  }

  entity "academy_profiles" as academy_profiles {
    *user_id : UUID <<PK, FK>>
    --
    academy_name : VARCHAR(120)
    ruc : VARCHAR(20) <<UQ>>
    verification_status : VARCHAR(20)
    website : VARCHAR(255)
  }

  entity "notification_preferences" as notification_preferences {
    *user_id : UUID <<PK, FK>>
    --
    email_enabled : BOOLEAN
    in_app_enabled : BOOLEAN
    community_enabled : BOOLEAN
    moderation_enabled : BOOLEAN
  }
}

    --
    *name : VARCHAR(80) <<UQ>>
  }

  entity "tags" as tags {
    *id : UUID <<PK>>
    --
    *name : VARCHAR(60) <<UQ>>
  }

    *id : UUID <<PK>>
    --
    *file_url : TEXT

  entity "academic_resources" as academic_resources {

    *author_user_id : UUID <<FK>>

  }

  entity "resource_tags" as resource_tags {
    *resource_id : UUID <<PK, FK>>

    created_at : TIMESTAMP
  }

  entity "download_logs" as download_logs {
    *id : UUID <<PK>>
    --
    *user_id : UUID <<FK>>
    *resource_id : UUID <<FK>>
    *downloaded_at : TIMESTAMP
  }
}

package "community" {
  entity "threads" as threads {
    *id : UUID <<PK>>
    --
    *author_user_id : UUID <<FK>>
    *subject_id : UUID <<FK>>
    *title : VARCHAR(160)
    *body : TEXT
    *is_anonymous : BOOLEAN
    *status : VARCHAR(20)
    created_at : TIMESTAMP
    updated_at : TIMESTAMP
  }

  entity "answers" as answers {
    *id : UUID <<PK>>
    --
    *thread_id : UUID <<FK>>
    *author_user_id : UUID <<FK>>
    *body : TEXT
    *is_accepted : BOOLEAN
    created_at : TIMESTAMP
    updated_at : TIMESTAMP
  }

  entity "comments" as comments {
    *id : UUID <<PK>>
    --
    *thread_id : UUID <<FK>>
    answer_id : UUID <<FK>>
    *author_user_id : UUID <<FK>>
    *body : TEXT
    created_at : TIMESTAMP
    updated_at : TIMESTAMP
  }

  entity "follow_relations" as follow_relations {
    *id : UUID <<PK>>
    --
    *follower_user_id : UUID <<FK>>
    *followed_user_id : UUID <<FK>>
    created_at : TIMESTAMP
  }

  entity "reactions" as reactions {
    *id : UUID <<PK>>
    --
    *user_id : UUID <<FK>>
    thread_id : UUID <<FK>>
    answer_id : UUID <<FK>>
    comment_id : UUID <<FK>>
    *reaction_type : VARCHAR(20)
    created_at : TIMESTAMP
  }
}

package "moderation" {
  entity "reports" as reports {
    *id : UUID <<PK>>

    *reporter_user_id : UUID <<FK>>

    *report_id : UUID <<FK>>

    created_at : TIMESTAMP
  }

  entity "audit_logs" as audit_logs {
    *id : UUID <<PK>>
    --
    *actor_user_id : UUID <<FK>>
    *action_type : VARCHAR(50)
    *entity_type : VARCHAR(50)
    *entity_id : UUID
    detail : TEXT
    created_at : TIMESTAMP
  }

  entity "appeals" as appeals {
    *id : UUID <<PK>>
    --
    *report_id : UUID <<FK>>
    *user_id : UUID <<FK>>
    *reason : TEXT
    *status : VARCHAR(20)
    created_at : TIMESTAMP
    reviewed_at : TIMESTAMP
  }
}
